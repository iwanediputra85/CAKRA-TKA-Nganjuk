Integrasi Database Google Sheets

Buka File Google Sheets
Penting untuk Deploy Netlify (Multi-Browser):
Agar semua Lembaga & Pengawas di HP/browser berbeda langsung terhubung tanpa setting ulang, tempelkan URL Web App pada variabel DEFAULT_APPS_SCRIPT_URL di dalam file index.html sebelum di-upload ke Netlify, atau klik tombol Salin Kode Siap Netlify di bawah.

Tempelkan link spreadsheet Google Sheets Anda agar tombol "Buka File Google Sheets" mengarah langsung ke file tersebut.

URL hasil deploy Apps Script agar seluruh isian Lembaga dan Pengawas otomatis tersimpan ke baris Google Sheets.

Langkah-Langkah Menghubungkan ke Google Sheets & Netlify

  1. Buka sheets.new untuk membuat Google Sheets baru dan beri nama Database CAKRA-TKA Nganjuk.
  2. Di dalam Google Sheets, klik menu Ekstensi (Extensions) → Apps Script.
  3. Hapus kode bawaan, lalu Salin & Tempel kode Apps Script terbaru di bawah ini ke editor Apps Script, lalu klik Simpan (Disket).
  4. Klik Terapkan (Deploy) → Deployment baru → Pilih Aplikasi Web → Jalankan sebagai: Saya (Me) → Yang memiliki akses: Siapa saja (Anyone).
  5. Salin URL Aplikasi Web (.../exec) dan tempelkan pada kolom 2. URL Web App Google Apps Script di samping, lalu klik Simpan Pengaturan Sheets.
  6. Klik tombol hijau "Unduh index.html Terkoneksi (Siap Netlify)", lalu unggah file index.html hasil unduhan tersebut ke Netlify agar seluruh browser/HP langsung terhubung!
Kode Google Apps Script (Versi Anti-CORS & Multi-Browser) 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetAcc = getOrCreateSheet(ss, "Accounts", ["username", "password", "name", "role", "kecamatan", "createdAt"]);
  var sheetRec = getOrCreateSheet(ss, "Records", ["npsn", "schoolName", "kecamatan", "scoresBind", "scoresMtk", "aiAnalysis", "checklist", "notes", "updatedAt"]);
  
  // Mendukung penyimpanan via parameter GET agar 100% bebas blokir CORS di Netlify & semua browser
  if (e && e.parameter && e.parameter.action) {
    var action = e.parameter.action;
    var payload = JSON.parse(e.parameter.payload || "{}");
    processWriteAction(sheetAcc, sheetRec, action, payload);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", action: action }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  var accounts = readSheetAsObjects(sheetAcc);
  var records = readSheetAsObjects(sheetRec);
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    accounts: accounts,
    records: records
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetAcc = getOrCreateSheet(ss, "Accounts", ["username", "password", "name", "role", "kecamatan", "createdAt"]);
  var sheetRec = getOrCreateSheet(ss, "Records", ["npsn", "schoolName", "kecamatan", "scoresBind", "scoresMtk", "aiAnalysis", "checklist", "notes", "updatedAt"]);
  var body = JSON.parse(e.postData.contents);
  
  processWriteAction(sheetAcc, sheetRec, body.action, body.data);
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function processWriteAction(sheetAcc, sheetRec, action, data) {
  if (action === "saveAccount") {
    var accObj = {
      username: "'" + String(data.username),
      password: "'" + String(data.password),
      name: String(data.name || ""),
      role: String(data.role || "lembaga"),
      kecamatan: String(data.kecamatan || ""),
      createdAt: String(data.createdAt || new Date().toISOString())
    };
    upsertRow(sheetAcc, "username", accObj, String(data.username));
  } else if (action === "saveRecord") {
    var rowObj = {
      npsn: "'" + String(data.npsn),
      schoolName: String(data.schoolName || ""),
      kecamatan: String(data.kecamatan || ""),
      scoresBind: JSON.stringify(data.scoresBind || {}),
      scoresMtk: JSON.stringify(data.scoresMtk || {}),
      aiAnalysis: JSON.stringify(data.aiAnalysis || null),
      checklist: JSON.stringify(data.checklist || []),
      notes: String(data.notes || ""),
      updatedAt: String(data.updatedAt || new Date().toISOString())
    };
    upsertRow(sheetRec, "npsn", rowObj, String(data.npsn));
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#991b1b").setFontColor("#ffffff");
  }
  return sheet;
}

function readSheetAsObjects(sheet) {
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      var val = data[i][j];
      if (headers[j] === "username" || headers[j] === "password" || headers[j] === "npsn") {
        val = String(val).replace(/^'/, "");
      } else if (typeof val === "string" && (val.startsWith("{") || val.startsWith("[") || val === "null")) {
        try { val = JSON.parse(val); } catch(err) {}
      }
      obj[headers[j]] = val;
    }
    result.push(obj);
  }
  return result;
}

function upsertRow(sheet, keyCol, obj, rawKey) {
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var keyIdx = headers.indexOf(keyCol);
  var rowIndex = -1;
  for (var i = 1; i < data.length; i++) {
    var cellVal = String(data[i][keyIdx]).replace(/^'/, "");
    if (cellVal === String(rawKey)) {
      rowIndex = i + 1;
      break;
    }
  }
  var rowValues = headers.map(function(h) { return obj[h] !== undefined ? obj[h] : ""; });
  if (rowIndex > 0) {
    sheet.getRange(rowIndex, 1, 1, headers.length).setValues([rowValues]);
  } else {
    sheet.appendRow(rowValues);
  }
}
